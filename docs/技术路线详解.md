- `wait_for` 2 分钟超时，防止 LLM 调用卡死

```
PricePredictor.execute()
    │
    ├── self.predict(material)
    │     │
    │     ├── self._predict_bayesian(material)    ← 优先
    │     │     │
    │     │     ├── 品类截距: alpha = alpha_cat[cat_idx] or mu_global
    │     │     ├── 供应商效应: beta = beta_supp[supp_idx] or 0
    │     │     ├── 数量折扣: gamma × (log(qty) - log_qty_mean)
    │     │     ├── mu_log = alpha + beta + qty_effect
    │     │     ├── pred_sd = sigma (已知供应商) or sqrt(sigma²+sigma_supp²)
    │     │     ├── P50 = exp(mu_log)
    │     │     ├── P10 = exp(mu_log - 1.28 × pred_sd)   ← 10%分位
    │     │     └── P90 = exp(mu_log + 1.28 × pred_sd)   ← 90%分位
    │     │
    │     └── self._predict_quantile(material)     ← 降级
    │           → 同品类分位数 + 数量系数
    │
    └── 返回 {p10, p50, p90, confidence, model}
```

### 偏离度打分内部：`AnomalyScorer`

```python
def calculate_deviation(self, quote, prediction, cost_analysis, material):
    category = material.get('category', '')
    cost_data_quality = cost_analysis.get('data_quality', 'reference_only')

    # ==== 权重选择（3种策略）====
    if category in ['塑料外壳', 'PCB板']:
        if cost_data_quality == 'reference_only':
            alpha, beta, gamma = 0.65, 0.05, 0.30   # 历史充足，成本仅参考
        else:
            alpha, beta, gamma = 0.50, 0.30, 0.20
    else:
        if cost_data_quality == 'reference_only':
            alpha, beta, gamma = 0.30, 0.05, 0.65   # 外部数据主导
        else:
            alpha, beta, gamma = 0.20, 0.20, 0.60

    # ==== 三个子分 ====
    # ① 价格偏离（基于预测 P50）
    price_deviation = abs(quote - pred_mid) / pred_mid * 100

    # ② 成本偏离（来自成本拆解）
    cost_deviation = cost_analysis.get('cost_deviation_score', 0)

    # ③ 市场偏离（外部 RAG 检索）
    rag_result = self._rag.calculate_external_deviation(quote, material)
    market_deviation = rag_result['external_deviation']

    # ==== 第一层：偏离度 ====
    deviation_score = (alpha * price_deviation
                     + beta * cost_deviation
                     + gamma * market_deviation)

    # ==== 第二层：综合打分（外部校准） ====
    composite_score = deviation_score * 0.6 + external_deviation * 0.4

    # ==== 严重级别 ====
    severity = '正常' if score < 20 else '关注' if score < 40 else '警示' if score < 60 else '紧急'

    return {
        'price_deviation': round(price_deviation, 1),
        'cost_deviation': round(cost_deviation, 1),
        'market_deviation': round(market_deviation, 1),
        'deviation_score': round(deviation_score, 1),
        'severity_level': severity,
        'composite_score': round(composite_score, 1),
        'weights': {'alpha': alpha, 'beta': beta, 'gamma': gamma},
        'rag': { ... },  # 外部参考区间
    }
```

```python
def _call_kimi_with_tools(system_prompt, messages, tools):
    api_key = os.environ.get("KIMI_API_KEY")
    base_url = os.environ.get("KIMI_BASE_URL", "https://ai-gateway.ailab.jiuan.com/v1")
    model = os.environ.get("KIMI_MODEL", "kimi-k2.5")

    if not api_key:
        return {"role": "assistant", "content": "KIMI_API_KEY not set", "tool_calls": []}

    client = OpenAI(api_key=api_key, base_url=base_url, timeout=60.0)
    full_messages = [{"role": "system", "content": system_prompt}] + list(messages)

    response = client.chat.completions.create(
        model=model,
        messages=full_messages,
        tools=tools,                    # ← function calling
        tool_choice="auto",
        temperature=0.3,
        max_tokens=2048,
    )

    return {
        "role": "assistant",
        "content": msg.content or "",
        "tool_calls": [{
            "id": tc.id,
            "function": {"name": tc.function.name, "arguments": tc.function.arguments}
        } for tc in (msg.tool_calls or [])],
    }
```

```python
def _compute_diff(old: Dict, new: Dict) -> Dict:
    diff = {}
    compare_fields = [
        "deviation_score", "severity_level", "phase",
        "ai_prediction_low", "ai_prediction_mid", "ai_prediction_high",
        "price_deviation", "cost_deviation", "market_deviation",
        "composite_score", "external_deviation",
    ]
    for field in compare_fields:
        old_val = safe(old.get(field))
        new_val = safe(new.get(field))
        if old_val is not None or new_val is not None:
            diff[field] = {"old": old_val, "new": new_val}
            if is_number(old_val) and is_number(new_val):
                diff[field]["change"] = round(new_val - old_val, 4)

    # 诊断结论对比
    if old.get("diagnosis_conclusion") or new.get("diagnosis_conclusion"):
        diff["diagnosis_conclusion"] = {"old": old.get(...), "new": new.get(...)}

    # 方案对比
    old_sols = old.get("solutions") or []
    new_sols = new.get("solutions") or []
    if len(old_sols) != len(new_sols) or old_sols != new_sols:
        diff["solutions"] = {"old": old_sols, "new": new_sols}

    return diff
```

```python
    compare_with = get_quote_by_id(conn, compare_id)
    diff = _compute_diff(original, compare_with)
    return {"original": original, "compare_with": compare_with, "diff": diff}
```
